第 2 步：建三张表（空留给你填）

    id   INTEGER PRIMARY KEY,
    ______
);

-- 第 2 张表：______
CREATE TABLE ______ (
    id   INTEGER PRIMARY KEY,
    ______
);

-- 第 3 张 = 绳子表：它的每一行 = 一条关系，不是一个人

    ______ INTEGER,      -- 指向第 ___ 张表
    ______ INTEGER,      -- 指向第 ___ 张表
    ______,              -- 还有别的要记吗？（想想 dosage 在哪）

);
